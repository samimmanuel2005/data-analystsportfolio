-- Query for aggregated sales & margin calculation
WITH RegionalAggregates AS (
  SELECT 
    category_id,
    SUM(order_total) AS total_revenue,
    AVG(order_total) AS avg_order_val,
    COUNT(DISTINCT customer_id) AS active_buyers
  FROM orders
  WHERE order_status = 'Completed'
  GROUP BY category_id
)
SELECT 
  r.category_id,
  r.total_revenue,
  ROUND(r.avg_order_val, 2) AS formatted_aov,
  DENSE_RANK() OVER (ORDER BY r.total_revenue DESC) AS rev_rank
FROM RegionalAggregates r;
